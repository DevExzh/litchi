```bash

export CARGO_PROFILE_DEV_DEBUG=0 CARGO_PROFILE_TEST_DEBUG=0 CARGO_TERM_COLOR=never

  phase=$1
  shift
  phase_root="$ROOT/$phase"

  export LITCHI_GATE_TARGET_ROOT="$CARGO_TARGET_DIR"
  export TMPDIR="$phase_root/tmp"
  mkdir "$TMPDIR"
  cargo "$@" 2>&1 | tee "$phase_root/command.log"
}
```
