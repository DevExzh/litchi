#!/usr/bin/env python3
"""Check retained binary/report identities and the unchanged oracle on pilots."""
import argparse
import hashlib
import importlib.util
import json
from pathlib import Path
import subprocess
import sys

ROOT = Path(__file__).resolve().parent
REPO = ROOT.parents[3]


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--role', choices=('before', 'after'), required=True)
    args = parser.parse_args()
    spec = importlib.util.spec_from_file_location('oracle440', ROOT / 'oracle/verify-report.py')
    oracle = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(oracle)
    identities = json.loads((ROOT / args.role / 'binary-copies.json').read_text())
    revision = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=REPO, text=True).strip()
    output = ROOT / 'pilots' / args.role
    output.mkdir(parents=True, exist_ok=False)
    for mode in ('normal', 'allocator'):
        identity = identities[mode]
        binary = Path(identity['path'])
        assert binary.stat().st_size == identity['bytes']
        assert hashlib.sha256(binary.read_bytes()).hexdigest() == identity['sha256']
        for shape in ('tiny', 'medium', 'large'):
            report = output / f'{mode}-{shape}.json'
            catalog = output / f'{mode}-{shape}-catalog.json'
            command = ['taskset', '-c', '2', str(binary), '--case', 'odp_existing_append_lifecycle', '--semantic-shape', shape, '--workers', '1', '--samples', '3', '--warmup', '1', '--json', str(report), '--corpus-manifest', str(catalog)]
            subprocess.run(command, cwd=REPO, check=True)
            value = json.loads(report.read_text())
            assert value['environment']['git_revision'] == revision
            assert value['tool']['binary']['sha256'] == identity['sha256']
            assert value['tool']['binary']['bytes'] == identity['bytes']
            assert value['tool']['binary']['path'] == str(binary)
            oracle.validate_report(report, mode, shape, samples=3, warmups=1)
            (output / f'{mode}-{shape}-receipt.json').write_text(json.dumps({'status':'pass', 'argv':command, 'binary':identity, 'revision':revision, 'oracle_sha256':hashlib.sha256((ROOT/'oracle/verify-report.py').read_bytes()).hexdigest()}, indent=2)+'\n')
    print('VALID: six pilots')


if __name__ == '__main__':
    main()
